```python
# ==================== 符号格式转换 ====================
def _jq_to_aq(code: str) -> str:
    """518880.XSHG -> 518880.SH，159915.XSHE -> 159915.SZ"""
    s = str(code).strip()
    s = s.replace('.XSHG', '.SH').replace('.XSHE', '.SZ')
    return adjust_symbol(s)


def _to_yyyymmdd(d) -> str:
    """date/datetime/str -> 'YYYYMMDD'"""
    if isinstance(d, datetime):
        return d.strftime('%Y%m%d')
    if isinstance(d, date):
        return d.strftime('%Y%m%d')
    return str(d).replace('-', '')[:8]
```

```python
    # ==================== 数据层私有方法 ====================

    def _ensure_trade_calendar(self) -> None:
        """从基准ETF历史日K中提取交易日历，缓存到 self._trade_calendar。"""
        if self._trade_calendar:
            return
        bt = self.config.get('backtest') or {}
        start = str(bt.get('start_date', self.config.get('backtest_start', '20100101')))
        end = str(bt.get('end_date', self.config.get('backtest_end', '20500101')))
        df = self.get_hist_data(self.risk_benchmark, period='D',
                                start_date=start, end_date=end)
        if df is None or df.empty:
            return
        dates = pd.to_datetime(df['date']).dt.date.tolist()
        self._trade_calendar = sorted(set(dates))

    def _get_trade_days(self, start_date=None, end_date=None,
                        count: Optional[int] = None) -> List[date]:
        """从缓存日历过滤，等价于原兼容层 _jq_get_trade_days。"""
        self._ensure_trade_calendar()
        days = list(self._trade_calendar)
        if end_date is not None:
            ed = end_date if isinstance(end_date, date) else \
                datetime.strptime(_to_yyyymmdd(end_date), '%Y%m%d').date()
            days = [x for x in days if x <= ed]
        if start_date is not None:
            sd = start_date if isinstance(start_date, date) else \
                datetime.strptime(_to_yyyymmdd(start_date), '%Y%m%d').date()
            days = [x for x in days if x >= sd]
        if count is not None and count > 0:
            days = days[-count:]
        return days

    def _previous_trade_date(self, ref: date) -> Optional[date]:
        """返回 ref 之前最近的交易日。"""
        self._ensure_trade_calendar()
        prev = [x for x in self._trade_calendar if x < ref]
        return prev[-1] if prev else None

    def _get_hist_batch(self, symbols: List[str], count: int,
                        end_date, fields: Optional[List[str]] = None) -> pd.DataFrame:
        """
        批量拉取多标的日K，返回含 (time, code, close, volume, money, high, low) 的 DataFrame。
        等价于原兼容层 _jq_get_price(syms, count=N, end_date=D, frequency='daily')。
        """
        end_d = _to_yyyymmdd(end_date)
        # 用 count 反推 start_date，留充足余量
        end_dt = datetime.strptime(end_d, '%Y%m%d')
        lookback_days = int(count * 1.8) + 60
        start_d = (end_dt - timedelta(days=lookback_days)).strftime('%Y%m%d')
        cache_key = (tuple(sorted(symbols)), start_d, end_d, count)
        if cache_key in self._hist_cache:
            return self._hist_cache[cache_key].copy()
        frames: List[pd.DataFrame] = []
        for sym in symbols:
            df = self.get_hist_data(sym, period='D', start_date=start_d, end_date=end_d)
            if df is None or df.empty:
                continue
            dfc = df.copy()
            dfc['time'] = pd.to_datetime(dfc['date'])
            dfc['code'] = sym
            # amount -> money
            if 'amount' in dfc.columns and 'money' not in dfc.columns:
                dfc['money'] = dfc['amount']
            elif 'money' not in dfc.columns:
                dfc['money'] = dfc.get('volume', 0) * dfc.get('close', 0)
            # 保留需要的列
            keep = ['time', 'code', 'close', 'volume', 'money', 'high', 'low']
            dfc = dfc[[c for c in keep if c in dfc.columns]]
            # 按 count 截尾
            dfc = dfc.sort_values('time').tail(count)
            frames.append(dfc)
        if not frames:
            return pd.DataFrame()
        out = pd.concat(frames, ignore_index=True)
        self._hist_cache[cache_key] = out
        return out.copy()

    def _get_hist_single(self, symbol: str, count: int, end_date,
                         period: str = 'D') -> pd.DataFrame:
        """单标的历史数据，返回含 (time, code, close, volume, money) 的 DataFrame。"""
        end_d = _to_yyyymmdd(end_date)
        end_dt = datetime.strptime(end_d, '%Y%m%d')
        lookback = int(count * 1.8) + 60
        start_d = (end_dt - timedelta(days=lookback)).strftime('%Y%m%d')
        df = self.get_hist_data(symbol, period=period, start_date=start_d, end_date=end_d)
        if df is None or df.empty:
            return pd.DataFrame()
        dfc = df.copy()
        dfc['time'] = pd.to_datetime(dfc['date'])
        dfc['code'] = symbol
        if 'amount' in dfc.columns and 'money' not in dfc.columns:
            dfc['money'] = dfc['amount']
        elif 'money' not in dfc.columns:
            dfc['money'] = dfc.get('volume', 0) * dfc.get('close', 0)
        keep = ['time', 'code', 'close', 'volume', 'money', 'high', 'low']
        dfc = dfc[[c for c in keep if c in dfc.columns]]
        return dfc.sort_values('time').tail(count).reset_index(drop=True)

    def _build_market_data(self, symbols: List[str]) -> _MarketDataMap:
        """
        为给定标的列表构建行情快照。
        实盘：调 get_realtime；回测：从历史数据取最后一根收盘价。
        等价于原兼容层 _jq_get_current_data()。
        """
        slices: Dict[str, _MarketData] = {}
        ctx = self._last_context
        end_d = _to_yyyymmdd(ctx.previous_date) if (ctx and ctx.previous_date) else \
            datetime.now().strftime('%Y%m%d')
        for sym in symbols:
            rt = self.get_realtime(sym) or {}
            lp = float(rt.get('最新价', 0) or 0)
            hi = float(rt.get('涨停', 0) or 0)
            lo = float(rt.get('跌停', 0) or 0)
            yc = float(rt.get('昨收', 0) or 0)
            # 回测或实盘无实时价格时降级到历史收盘
            if lp <= 0:
                try:
                    hdf = self.get_hist_data(sym, period='D', start_date=end_d, end_date=end_d)
                    if hdf is not None and not hdf.empty and 'close' in hdf.columns:
                        lp = float(hdf['close'].iloc[-1])
                        if len(hdf) >= 2:
                            yc = float(hdf['close'].iloc[-2])
                        elif yc <= 0:
                            yc = lp
                except Exception:
                    pass
            if hi <= 0 and yc > 0:
                hi = yc * 1.1
            if lo <= 0 and yc > 0:
                lo = yc * 0.9
            slices[sym] = _MarketData(
                last_price=lp, high_limit=hi, low_limit=lo, paused=(lp <= 0)
            )
        return _MarketDataMap(slices)

    def _build_context(self, event_time: Optional[datetime] = None) -> _Context:
        """构建策略上下文快照，等价于原兼容层 build_context()。"""
        now = event_time or datetime.now()
        prev = self._previous_trade_date(now.date())
        positions: Dict[str, _PositionView] = {}
        all_pos = self.get_all_positions() or {}
        for sym, pos in all_pos.items():
            sym_aq = adjust_symbol(sym)
            vol = float(getattr(pos, 'volume', 0) or 0)
            if vol <= 0:
                continue
            avail = float(getattr(pos, 'available', vol) or vol)
            avg = float(getattr(pos, 'avg_price', 0) or 0)
            px = float(getattr(pos, 'current_price', 0) or 0) or avg
            positions[sym_aq] = _PositionView(
                total_amount=vol, closeable_amount=avail, avg_cost=avg, price=px
            )
        cash = float(self.position_manager.cash)
        tv = float(self.position_manager.total_assets)
        # 允许外部注入真实账户数据（实盘用）
        acc_fn = self.config.get('external_get_account')
        if callable(acc_fn):
            acc = acc_fn() or {}
            if acc.get('cash') is not None:
                cash = float(acc['cash'])
            if acc.get('total_value') is not None:
                tv = float(acc['total_value'])
        return _Context(
            current_dt=now,
            previous_date=prev,
            portfolio=_PortfolioView(positions=positions,
                                     available_cash=cash, total_value=tv),
        )

    def _get_all_securities_df(self) -> pd.DataFrame:
        """
        获取全市场ETF列表，返回含 display_name 列、以AQ代码为index的DataFrame。
        等价于原兼容层 _jq_get_all_securities(['etf'])。
        """
        df = self.data_engine.get_etf_spot()
        if df is None or df.empty:
            return pd.DataFrame(columns=['display_name'])
        # get_etf_spot 返回 '代码' '名称' 等列
        code_col = '代码' if '代码' in df.columns else df.columns[0]
        name_col = '名称' if '名称' in df.columns else df.columns[1]
        out = pd.DataFrame(index=df[code_col].astype(str).map(adjust_symbol))
        out['display_name'] = df[name_col].values
        return out

    def _clear_day_cache(self) -> None:
        """每日清空历史数据缓存和行情快照缓存。"""
        self._hist_cache.clear()
        self._market_data_cache = None
```

```python
    def calculate_premium_rate(self, etf: str, context: _Context):
        """
        计算ETF溢价率。enable_premium_filter=False 时快速返回 (None, True)。

            passed_premium = float(premium_rate) <= self.max_premium_rate
            return float(premium_rate), passed_premium
        except Exception as e:
```
